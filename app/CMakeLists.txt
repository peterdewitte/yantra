# The desktop application (OA-17, UI-13). Only built with YANTRA_BUILD_APP=ON;
# the GUI libraries (Dear ImGui, GLFW, OpenGL) are found here and nowhere
# else, so a build with YANTRA_BUILD_APP=OFF needs none of them.
# docs/APP.md describes the structure:
#
#   yantra_app_model      state, actions, settings, command line, window-system
#                         choice, data folders, icons: no GUI library, tested by
#                         tests/app (TST-50)
#   yantra_imgui          Dear ImGui core (third_party/imgui) with Yantra's item
#                         probe (app/gui/ui_probe.cpp, IMGUI_ENABLE_TEST_ENGINE
#                         hooks): no window system or OpenGL
#   yantra_app_gui        the ImGui layer (app/gui): needs an ImGui context only,
#                         so the headless UI tests draw it as the app does
#   yantra_imgui_backends Dear ImGui's GLFW and OpenGL 3 backends
#   yantra                the executable (app/src/main.cpp): window, GL, loop
# Copied from yantra-v2 68ea20a: app/CMakeLists.txt, adapted (headless GUI
# library and probe).

# UI-12: the language files data/i18n/<lang>.json are built into the
# model library (i18n.h), as pieces of raw text (a single literal would
# pass the ISO C++ length limit); editing one reconfigures.
set(_i18n_cpp "// Generated by app/CMakeLists.txt from data/i18n/*.json: do not edit.\n#include <cstddef>\nnamespace yantra::app::i18n_data {\n")
foreach(_lang nl fr de)
  set(_file ${PROJECT_SOURCE_DIR}/data/i18n/${_lang}.json)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_file})
  file(READ ${_file} _text)
  string(LENGTH "${_text}" _len)
  string(TOUPPER ${_lang} _up)
  string(SUBSTRING ${_up} 0 1 _first)
  string(SUBSTRING ${_lang} 1 -1 _rest)
  set(_name "k${_first}${_rest}")
  string(APPEND _i18n_cpp "extern const char* const ${_name}[];\nextern const std::size_t ${_name}Count;\nconst char* const ${_name}[] = {\n")
  set(_pos 0)
  set(_count 0)
  while(_pos LESS _len)
    string(SUBSTRING "${_text}" ${_pos} 12000 _piece)
    string(APPEND _i18n_cpp "R\"yantra_i18n(${_piece})yantra_i18n\",\n")
    math(EXPR _pos "${_pos} + 12000")
    math(EXPR _count "${_count} + 1")
  endwhile()
  if(_count EQUAL 0)
    string(APPEND _i18n_cpp "\"\",\n")
    set(_count 1)
  endif()
  string(APPEND _i18n_cpp "};\nconst std::size_t ${_name}Count = ${_count};\n")
endforeach()
string(APPEND _i18n_cpp "}  // namespace yantra::app::i18n_data\n")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/i18n_tables.cpp CONTENT "${_i18n_cpp}" @ONLY)
